    fn redact_paths(value: &mut serde_json::Value, workspace: &str) {

                if s.contains(workspace) || s.starts_with('/') {
                    let trimmed = s.trim_start_matches(workspace).trim_start_matches('/').to_owned();

                }
            }
            // recurse into arrays and objects...
        }
    }
